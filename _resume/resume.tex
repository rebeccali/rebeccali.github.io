% One bold line, right-aligned trailer.
\newcommand{\resumeSubheadingOnly}[2]{
	\vspace{-1pt}\item[]
	\begin{tabular*}{0.97\textwidth}{l@{\extracolsep{\fill}}r}
		\textbf{#1} & #2 \\
	\end{tabular*}\vspace{-2pt}
}

% Bold line with a right-aligned location, then a wrapped body paragraph
% with the dates set against it on the right.
\newcommand{\resumeSubheadingCleanLong}[4]{
	\vspace{-1pt}\item[]
	\begin{tabular*}{0.97\textwidth}{L@{\extracolsep{\fill}}r}
		\textbf{#1} & {\small #2} \\
		\textit{\small #4} &  \textit{\small #3} \\
	\end{tabular*}\vspace{-0pt}

	% The two paragraph breaks below are load-bearing: they set the gap
	% between employment entries. Removing them tightens the whole section.

}

% Bold term, then a run-on description.
\newcommand{\resumeItem}[2]{
	\item[]\small{
		\textbf{#1}{: #2 \vspace{-2pt}}
	}
}
\newcommand{\resumeSubItem}[2]{\resumeItem{#1}{#2}\vspace{-5pt}}

% A line with a right-aligned date, for loose lists.
\newcommand{\itemAndDate}[2]{
	\vspace{-1pt}\item[]
	\begin{tabular*}{0.97\textwidth}{l@{\extracolsep{\fill}}r}
		#1 & \textit{#2} \\
	\end{tabular*}\vspace{-8pt}
}
